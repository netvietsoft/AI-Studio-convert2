// SMPTE 240M transfer functions (per ITU-R BT.2380-1, Table 2.7, transfer_characteristic = 7).
// Differs from BT.601 only in the linear-segment slope (4.0 vs 4.5);
// threshold 0.018, gain 1.099, offset 0.099, gamma 0.45 are the same as BT.601.
// EOTF break point is 4.0 * 0.018 = 0.072 (vs 0.081 for BT.601).
vec4 SMPTE240M_OETF(vec4 rgb)
    rgb.x = clamp(rgb.x < 0.018 ? rgb.x * 4.0 : 1.099 * pow(abs(rgb.x), 0.45) - 0.099, 0.0, 1.0);
    rgb.y = clamp(rgb.y < 0.018 ? rgb.y * 4.0 : 1.099 * pow(abs(rgb.y), 0.45) - 0.099, 0.0, 1.0);
    rgb.z = clamp(rgb.z < 0.018 ? rgb.z * 4.0 : 1.099 * pow(abs(rgb.z), 0.45) - 0.099, 0.0, 1.0);
    return rgb;
vec4 SMPTE240M_EOTF(vec4 rgb)
    rgb.x = clamp(rgb.x < 0.072 ? rgb.x / 4.0 : pow(abs((rgb.x + 0.099) / 1.099), 1.0 / 0.45), 0.0, 1.0);

    //rgb.w = rgb.w;
    return rgb;

float HLGOETF(float x)
    if (x > 1.0 / 12.)
    {
        return 0.17883277 * log(12. * x - 0.28466892) + 0.55991073;
    }
    return sqrt(max(0.0, 3.0 * x));
vec4 HLG_OETF(vec4 rgb)
    vec4 d;
    d.x = HLGOETF(rgb.x);
    d.y = HLGOETF(rgb.y);
    d.z = HLGOETF(rgb.z);
    d.w = rgb.w;
    return d;
float HLGInverseOETF(float x)
    if (x > 1.0 / 2.)
    {
        return (exp((x - 0.55991073) / 0.17883277) + 0.28466892) / 12.;
    }
    return pow(x, 2.0) / 3.0;

vec4 BT2446_Method_A_TMO(vec4 src_rgba)
    // --- 
 BT.2446 
) ---
    // c0: {0.0, 1/2.4, 0.678, 0.2627}
    // c1: {12.25979805, 1.0, 0.0593, 0.2681693435}
    // c2: {0.7399, -1.151, 0.9909, 1.077}
    // c3: {2.7811, 0.0, -0.6302, 0.5}
    // c4: {5.696958065, 1.1, 5.960464478e-08, 0.2129037529}
    // c5: {0.67815, 0.1, 0.5315190554, 0.0}
    vec4 dst_rgba;
    // 1. 
 (Gamma 1/2.4 
    float r_prime = max(src_rgba.x, 0.0);
    float g_prime = max(src_rgba.y, 0.0);
    float b_prime = max(src_rgba.z, 0.0);
    // R' = R^(1/2.4), G' = G^(1/2.4), B' = B^(1/2.4)
    r_prime = pow(r_prime, 0.4166666567);
    g_prime = pow(g_prime, 0.4166666567);
    b_prime = pow(b_prime, 0.4166666567);
    // 2. 
 (Rec.2020 
    // Y = 0.2627*R' + 0.6780*G' + 0.0593*B'
    float y_linear_part = r_prime * 0.2626999915 + g_prime * 0.6779999733;
    float y_linear = b_prime * 0.05930000171 + y_linear_part;
    // 3. 
    float y_log = log2(y_linear * 12.25979805 + 1.0) * 0.2681693435;
    // 4. BT.2446 Method A 
    float final_y_log;
    if (y_log > 0.9908999801) {
        // 
        final_y_log = y_log * 0.5 + 0.5;
    } else if (y_log > 0.7398999929) {
        // 
        final_y_log = y_log * (y_log * (-1.151000023) + 2.781100035) + (-0.6302000284);
    } else {
        // 
        final_y_log = y_log * 1.077000022 + 0.0;
    }
    // 5. 
    float compressed_y = pow(5.696958065, final_y_log) * 0.2129037529 - 0.2129037529;
    float denom = max(y_linear * 1.100000024, 5.960464478e-08);
    float gain = compressed_y / denom;
    // 6. 
    float r_diff_gain = (r_prime - y_linear) * gain;
    float b_diff_gain = (b_prime - y_linear) * gain;
    float scaled_r_diff = r_diff_gain * 0.6781499982;
    float scaled_b_diff = b_diff_gain * 0.5315190554;
    // 
    float y_adjust = max(scaled_r_diff * 0.1000000015, 0.0);
    // 7. YCbCr (BT.2020 NCL) 
    float Y  = compressed_y - y_adjust;
    float Cb = scaled_b_diff;
    float Cr = scaled_r_diff;
    dst_rgba.x = Y +                1.4746 * Cr;
    dst_rgba.y = Y - 0.1645 * Cb - 0.5713 * Cr;
    dst_rgba.z = Y + 1.8814 * Cb;
    dst_rgba.w = src_rgba.w;
    // 8. EOTF (Gamma 2.4 
    dst_rgba.x = clamp(pow(max(dst_rgba.x, 0.0), 2.4), 0.0, 1.0);
    dst_rgba.y = clamp(pow(max(dst_rgba.y, 0.0), 2.4), 0.0, 1.0);
    dst_rgba.z = clamp(pow(max(dst_rgba.z, 0.0), 2.4), 0.0, 1.0);
    return dst_rgba;
